# 待办事项

> 此文件为待办跟进主载体（Tasks 语法），项目唯一任务写入点。已完成项在使用者说「归档」时迁移至 `已完成归档.md`（**手动，不自动迁**）。
> 条线清单：综合事务/财务/法律/业务（本清单可按项目改；**AI 不判线，条线由使用者口头指定**）。
> 结构：`## 条线` 一级分组 → `### 负责人` 二级分组 → 任务写在对应人名标题下（负责人不写进任务正文）；无负责人任务直接放条线下。
> 格式：`- [ ] 事项 📅 YYYY-MM-DD ⏫`（日期／优先级均可选、不确定就不带；优先级标记 `🔺`最高／`⏫`高／`🔼`中／`🔽`低／`⏬`最低）。
> 加待办走「模板流」：对 AI 说「要加待办」→ AI 回模板 → 你改完回传 → AI 校验（条线在清单内／负责人有同名视图／日期合法）后写入。导出：`scripts/export_todo.py`（只导未完成，落下载目录；发不发由人）。

## 📋 待办总表（交互视图 · 只读，改动请改下方分区）

> **筛选**：条线／负责人下拉 ＋ 事项关键词搜索；**排序**：点列头切换（默认逾期置顶＋日期升序）；另有「只看逾期／只看未填截止」开关，一键「✕ 清除筛选」。**下方分区才是源数据**，改待办去分区里改。

```dataviewjs
const page = dv.current();
const TODAY = dv.date('today');
const todayStr = TODAY.toISODate();
const rows = [];
// 标题行号表：dv.io.load 读原文逐行解析（不依赖 dataview 的 sections/lists API，版本间行为不一致）
const headers = [];
{
    const src = (await dv.io.load(page.file.path)) || "";
    const ls = src.split("\n");
    for (let i = 0; i < ls.length; i++) {
        const m = ls[i].match(/^(#{2,3})\s+(.+)$/);
        if (m) headers.push({ line: i, level: m[1].length, name: m[2].trim() });
    }
}
for (const t of page.file.tasks) {
    if (t.completed) continue;
    let person = "", lineName = "";
    // 行号区间归属：任务按其行号落入最近的前置 ### （负责人）/ ## （条线）标题
    for (const h of headers) {
        if (h.line < t.line) {
            if (h.level === 3) person = h.name;
            else if (h.level === 2 && h.name.indexOf("待办总表") < 0) lineName = h.name;
        }
    }
    const s = t.text;
    const dm = s.match(/📅\s*(\d{4}-\d{2}-\d{2})/);
    const due = dm ? dm[1] : "";
    const od = due !== "" && dv.compare(dv.date(due), TODAY) < 0;
    rows.push({ 条线: lineName || "未分类", 负责人: person || "（未分派）",
        事项: s.replace(/📅\s*\d{4}-\d{2}-\d{2}/, "").trim(),
        截止日期: due || "—",
        状态: od ? "⚠ 逾期" : (due === todayStr ? "今日到期" : "进行中"),
        _od: od, _due: due || "9999-99-99" });
}

const uniq = (a) => Array.from(new Set(a)).sort((x, y) => x.localeCompare(y, "zh"));
const total = rows.length, totalOD = rows.filter(r => r._od).length;

let sortKey = "date", sortAsc = true, onlyOD = false, onlyNoDue = false;
let fLine = "", fPerson = "", fKw = "";

const root = dv.el("div", "");
const bar = document.createElement("div");
bar.style.cssText = "margin:6px 0 8px 0;display:flex;gap:6px;flex-wrap:wrap;align-items:center;";
const wrap = document.createElement("div");
root.appendChild(bar); root.appendChild(wrap);

const CTL = "font-size:12px;padding:2px 8px;border:1px solid #d0d7e2;border-radius:6px;background:transparent;color:inherit;";

// 控件只建一次（避免重绘时输入框失焦）
function mkSelect(options, allLabel, onChange) {
    const sel = document.createElement("select");
    sel.style.cssText = CTL;
    const o0 = document.createElement("option");
    o0.value = ""; o0.textContent = allLabel; sel.appendChild(o0);
    for (const v of options) {
        const o = document.createElement("option");
        o.value = v; o.textContent = v; sel.appendChild(o);
    }
    sel.onchange = () => onChange(sel.value);
    bar.appendChild(sel);
    return sel;
}
const selLine = mkSelect(uniq(rows.map(r => r.条线)), "全部条线", v => { fLine = v; render(); });
const selPerson = mkSelect(uniq(rows.map(r => r.负责人)), "全部负责人", v => { fPerson = v; render(); });

const inp = document.createElement("input");
inp.type = "text";
inp.placeholder = "搜索事项…";
inp.style.cssText = CTL + "min-width:150px;";
inp.oninput = () => { fKw = inp.value; render(); };
bar.appendChild(inp);

function render() {
    let rs = rows.slice();
    if (onlyOD) rs = rs.filter(r => r._od);
    if (onlyNoDue) rs = rs.filter(r => r.截止日期 === "—");
    if (fLine) rs = rs.filter(r => r.条线 === fLine);
    if (fPerson) rs = rs.filter(r => r.负责人 === fPerson);
    const kw = fKw.trim();
    if (kw) rs = rs.filter(r => r.事项.indexOf(kw) >= 0);
    rs.sort((a, b) => sortKey === "date"
        ? ((b._od ? 1 : 0) - (a._od ? 1 : 0)) || (a._due < b._due ? -1 : 1)
        : String(a[sortKey]).localeCompare(String(b[sortKey]), "zh") * (sortAsc ? 1 : -1));

    wrap.innerHTML = "";
    if (total === 0) {
        const empty = document.createElement("div");
        empty.style.cssText = "font-size:12px;color:#808080;padding:4px 0;";
        empty.textContent = "（本项目暂无未完成待办）";
        wrap.appendChild(empty);
        return;
    }
    const tbar = document.createElement("div");
    tbar.style.cssText = "margin:2px 0 6px 0;display:flex;gap:6px;flex-wrap:wrap;align-items:center;";
    const mkBtn = (label, act, active) => {
        const b = document.createElement("button");
        b.textContent = label;
        b.style.cssText = "font-size:12px;padding:2px 10px;border-radius:6px;cursor:pointer;" +
            (active ? "background:#2f5597;color:#fff;border:none;"
                    : "border:1px solid #d0d7e2;background:transparent;color:inherit;");
        b.onclick = act; tbar.appendChild(b); return b;
    };
    mkBtn("只看逾期", () => { onlyOD = !onlyOD; render(); }, onlyOD);
    mkBtn("只看未填截止", () => { onlyNoDue = !onlyNoDue; render(); }, onlyNoDue);
    if (fLine || fPerson || kw || onlyOD || onlyNoDue) {
        mkBtn("✕ 清除筛选", () => {
            fLine = ""; fPerson = ""; fKw = ""; onlyOD = false; onlyNoDue = false;
            selLine.value = ""; selPerson.value = ""; inp.value = "";
            render();
        }, false);
    }
    wrap.appendChild(tbar);

    const tb = document.createElement("table");
    tb.style.cssText = "width:100%;border-collapse:collapse;font-size:13px;";
    const thead = document.createElement("thead");
    const hr = document.createElement("tr");
    for (const [label, key] of [["条线","条线"],["负责人","负责人"],["事项","事项"],["截止日期","date"],["状态","状态"]]) {
        const th = document.createElement("th");
        th.textContent = label + (sortKey === key ? (sortAsc ? " ▲" : " ▼") : "");
        th.style.cssText = "text-align:left;padding:6px 8px;background:#2f5597;color:#fff;cursor:pointer;white-space:nowrap;";
        th.onclick = () => {
            if (key === "date") { sortKey = "date"; sortAsc = true; }
            else if (sortKey === key) { sortAsc = !sortAsc; } else { sortKey = key; sortAsc = true; }
            render();
        };
        hr.appendChild(th);
    }
    thead.appendChild(hr); tb.appendChild(thead);
    const tbody = document.createElement("tbody");
    rs.forEach((r, i) => {
        const tr = document.createElement("tr");
        tr.style.cssText = r._od ? "background:#fce4e4;color:#c00000;font-weight:600;"
            : (i % 2 ? "background:#f2f6fc;" : "");
        for (const k of ["条线", "负责人", "事项", "截止日期", "状态"]) {
            const td = document.createElement("td");
            td.textContent = r[k];
            td.style.cssText = "padding:5px 8px;border-bottom:1px solid #e5e7eb;vertical-align:top;" +
                (k === "事项" ? "" : "white-space:nowrap;");
            tr.appendChild(td);
        }
        tbody.appendChild(tr);
    });
    tb.appendChild(tbody); wrap.appendChild(tb);

    const foot = document.createElement("div");
    foot.style.cssText = "margin-top:6px;font-size:12px;color:#808080;";
    foot.textContent = "显示 " + rs.length + " / 共 " + total + " 条（逾期 " + totalOD + "）｜只读视图：改待办请改下方分区";
    wrap.appendChild(foot);
}
render();
```
